### 方法 4：快速统计指令（最灵活）

```powershell
cd C:\Users\User\ahpal-static

# 文章总数（排除游戏、分类页）
$count = (Get-ChildItem -Recurse -Filter "*.html" | Where-Object { $_.DirectoryName -notmatch "game" -and $_.Name -notmatch "index|categories|dashboard|about|contact|privacy|terms|404|memorial" }).Count
Write-Host "📊 文章總數：$count 篇" -ForegroundColor Cyan

# 各分类数量
foreach ($cat in @("tech","life","review","philosophy","trend")) {
    $c = (Get-ChildItem -Path $cat -Filter "*.html" -ErrorAction SilentlyContinue).Count
    Write-Host "   $cat ：$c 篇" -ForegroundColor Gray
}
Write-Host "   🎮 game ：$(Get-ChildItem game -Filter *.html -ErrorAction SilentlyContinue | Measure-Object | Select-Object -ExpandProperty Count) 款" -ForegroundColor Gray

# 异常检查
$bad = Get-ChildItem -Recurse -Filter "*.html" | Where-Object { $_.DirectoryName -notmatch "game" -and $_.Length -lt 5120 }
if ($bad) { 
    Write-Host "`n⚠️ 異常小檔案：$($bad.Count) 個" -ForegroundColor Yellow
    $bad | ForEach-Object { Write-Host "   $($_.Name) ($([math]::Round($_.Length/1KB,2)) KB)" -ForegroundColor Gray }
} else {
    Write-Host "`n✅ 無異常小檔案" -ForegroundColor Green
}

# 最新 5 篇
Write-Host "`n📅 最新 5 篇文章：" -ForegroundColor Cyan
Get-ChildItem -Recurse -Filter "*.html" | Where-Object { $_.DirectoryName -notmatch "game" -and $_.Name -notmatch "index|categories" } | Sort-Object LastWriteTime -Descending | Select-Object -First 5 | ForEach-Object { Write-Host "   $($_.LastWriteTime.ToString('yyyy-MM-dd HH:mm')) - $($_.Name)" -ForegroundColor Gray }
```
